from datetime import datetime, timedelta from typing import Optional from fastapi import FastAPI, HTTPException, Depends from pydantic import BaseModel, EmailStr from sqlalchemy import create_engine, Column, Integer, String, DateTime, Float, ForeignKey from sqlalchemy.ext.declarative import declarative_base from sqlalchemy.orm import sessionmaker, Session, relationship # إعداد قاعدة البيانات السحابية/المحلية باستخدام SQLite SQLALCHEMY_DATABASE_URL = "sqlite:///./black_iris_ai.db" engine = create_engine(SQLALCHEMY_DATABASE_URL, connect_args={"check_same_thread": False}) SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine) Base = declarative_base() # ==================== 1. نماذج قاعدة البيانات (Database Models) ==================== class UserModel(Base): __tablename__ = "users" id = Column(Integer, primary_key=True, index=True) email = Column(String, unique=True, index=True) signup_date = Column(DateTime, default=datetime.utcnow) subscription_status = Column(String, default="Trial") # Trial, Active, Expired selected_plan = Column(String, default="Free_15_Days") # Starter, Pro, Enterprise, Free_15_Days campaigns = relationship("CampaignModel", back_populates="owner") class CampaignModel(Base): __tablename__ = "campaigns" id = Column(Integer, primary_key=True, index=True) product_description = Column(String) target_platform = Column(String) ai_ad_copy = Column(String) budget = Column(Float) ad_status = Column(String, default="Active") # Active, Paused, Completed owner_id = Column(Integer, ForeignKey("users.id")) owner = relationship("UserModel", back_populates="campaigns") Base.metadata.create_all(bind=engine) # ==================== 2. هياكل المدخلات (Pydantic Schemas) ==================== class UserCreate(BaseModel): email: EmailStr class CampaignCreate(BaseModel): user_email: EmailStr product_description: str budget: float class PlanUpgrade(BaseModel): email: EmailStr plan_name: str # دالة الاتصال بقاعدة البيانات def get_db(): db = SessionLocal() try: yield db finally: db.close() # إعداد تطبيق FastAPI app = FastAPI( title="Black Iris AI Backend", description="Backend API for Black Iris AI Autonomous Advertising Platform", version="1.0.0" ) # ==================== 3. المسارات البرمجية للمنصة (API Endpoints) ==================== @app.post("/api/register") def register_user(user: UserCreate, db: Session = Depends(get_db)): """تسجيل مستخدم جديد ومنحه فترة تجريبية مجانية لمدة 15 يوماً تلقائياً""" db_user = db.query(UserModel).filter(UserModel.email == user.email).first() if db_user: return {"message": "المستخدم مسجل مسبقاً", "status": db_user.subscription_status} new_user = UserModel( email=user.email, signup_date=datetime.utcnow(), subscription_status="Trial", selected_plan="Free_15_Days" ) db.add(new_user) db.commit() db.refresh(new_user) return { "success": True, "message": "تم التسجيل بنجاح، وممنوح 15 يوماً تجريبية للمنصة", "user_id": new_user.id, "signup_date": new_user.signup_date } @app.get("/api/check-subscription") def check_subscription(email: str, db: Session = Depends(get_db)): """التحقق التلقائي من حالة الـ 15 يوماً التجريبية وتحديثها""" user = db.query(UserModel).filter(UserModel.email == email).first() if not user: raise HTTPException(status_code=404, detail="المستخدم غير موجود") if user.subscription_status == "Trial": days_passed = (datetime.utcnow() - user.signup_date).days if days_passed > 15: user.subscription_status = "Expired" db.commit() return { "status": "Expired", "message": "انتهت فترة التجربة المجانية (15 يوماً)، يرجى الترقية إلى إحدى الباقات الشهرية." } else: remaining_days = 15 - days_passed return { "status": "Trial", "remaining_days": remaining_days, "message": f"أنت في فترة التجربة المجانية، متبقي {remaining_days} يوم." } return { "status": user.subscription_status, "plan": user.selected_plan, "message": "الحساب نشط." } @app.post("/api/generate-and-launch-campaign") def generate_and_launch_campaign(data: CampaignCreate, db: Session = Depends(get_db)): """محاكاة عمل الذكاء الاصطناعي وإطلاق الحملة بعد التحقق من اشتراك المستخدم""" user = db.query(UserModel).filter(UserModel.email == data.user_email).first() if not user: raise HTTPException(status_code=404, detail="المستخدم غير مسجل") # التحقق من صلاحية التجربة أو الاشتراك if user.subscription_status == "Trial": if (datetime.utcnow() - user.signup_date).days > 15: user.subscription_status = "Expired" db.commit() raise HTTPException(status_code=403, detail="انتهت فترة التجربة المجانية، يرجى ترقية حسابك.") elif user.subscription_status == "Expired": raise HTTPException(status_code=403, detail="اشتراكك منتهي، يرجى الدفع لمتابعة إطلاق الحملات.") # تحليل الذكاء الاصطناعي لاختيار المنصة والنص التسويقي بشكل مرتب ومنسق desc_lower = data.product_description.lower() if "عطور" in desc_lower or "ملابس" in desc_lower or "مجوهرات" in desc_lower: platform = "Instagram & Snapchat" ad_copy = f"✨ تألق بإطلالة استثنائية مع {data.product_description}. اطلب قطعتك الآن!" elif "دورة" in desc_lower or "خدمة" in desc_lower: platform = "Google Search" ad_copy = f"💡 طور مهاراتك اليوم مع {data.product_description}. سجل الآن واحصل على استشارة مجانية." else: platform = "TikTok" ad_copy = f"🔥 تريند اليوم: لا تفوت فرصة تجربة {data.product_description}! اطلب الآن." # حفظ الحملة وإطلاقها new_campaign = CampaignModel( product_description=data.product_description, target_platform=platform, ai_ad_copy=ad_copy, budget=data.budget, ad_status="Active", owner_id=user.id ) db.add(new_campaign) db.commit() db.refresh(new_campaign) return { "success": True, "message": "تم تحليل المنتج وإطلاق الحملة بنجاح عبر المنصة المختصة!", "campaign_details": { "id": new_campaign.id, "platform": platform, "generated_copy": ad_copy, "budget": data.budget, "status": new_campaign.ad_status } } @app.post("/api/upgrade-subscription") def upgrade_subscription(data: PlanUpgrade, db: Session = Depends(get_db)): """ترقية الحساب إلى إحدى الباقات الشهرية (Starter, Pro, Enterprise)""" user = db.query(UserModel).filter(UserModel.email == data.email).first() if not user: raise HTTPException(status_code=404, detail="المستخدم غير موجود") valid_plans = ["Starter", "Pro", "Enterprise"] if data.plan_name not in valid_plans: raise HTTPException(status_code=400, detail="اسم الباقة غير صحيح") user.selected_plan = data.plan_name user.subscription_status = "Active" db.commit() return {"success": True, "message": f"تم تفعيل اشتراكك بنجاح في الباقة: {data.plan_name}"}
